    <script>
        const API_BASE = 'http://localhost:8080/api';
        let currentUser = null;
        let extractedResumeData = null;

        // Tab Navigation
        function switchTab(tabName) {
            document.querySelectorAll('.tab-content').forEach(tab => {
                tab.classList.remove('active');
            });
            document.querySelectorAll('.tab-btn').forEach(btn => {
                btn.classList.remove('active');
            });
            
            document.getElementById(tabName).classList.add('active');
            event.target.classList.add('active');
        }

        // Alert Handler
        function showAlert(elementId, message, type = 'info') {
            const alert = document.getElementById(elementId);
            alert.innerHTML = `<div class="alert alert-${type}">${message}</div>`;
            setTimeout(() => alert.innerHTML = '', 5000);
        }

        // ===== PROFILE TAB =====
        function registerUser(event) {
            event.preventDefault();

            if (!currentUser) {
                const userData = {
                    firstName: document.getElementById('firstName').value,
                    lastName: document.getElementById('lastName').value,
                    email: document.getElementById('email').value,
                    phone: document.getElementById('phone').value,
                    location: document.getElementById('location').value,
                    yearsExperience: document.getElementById('yearsExperience').value,
                    skills: document.getElementById('skills').value,
                    jobTitle: document.getElementById('jobTitle').value
                };

                fetch(`${API_BASE}/users/register`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(userData)
                })
                .then(res => {
                    if (!res.ok) throw new Error('Registration failed');
                    return res.json();
                })
                .then(data => {
                    currentUser = data;
                    localStorage.setItem('currentUser', JSON.stringify(data));
                    showAlert('profileAlert', 'Profile registered successfully!', 'success');
                    displayProfile();
                })
                .catch(err => showAlert('profileAlert', 'Error: ' + err.message, 'error'));
            }
        }

        function displayProfile() {
            if (currentUser) {
                document.getElementById('profileForm').classList.add('hidden');
                document.getElementById('profileDisplay').classList.remove('hidden');
                
                document.getElementById('displayName').textContent = `${currentUser.firstName} ${currentUser.lastName}`;
                document.getElementById('displayEmail').textContent = currentUser.email;
                document.getElementById('displayPhone').textContent = currentUser.phone;
                document.getElementById('displayLocation').textContent = currentUser.location;
                document.getElementById('displayExperience').textContent = currentUser.yearsExperience;
                document.getElementById('displaySkills').textContent = currentUser.skills;
                document.getElementById('displayJobTitle').textContent = currentUser.jobTitle;
            }
        }

        function editProfile() {
            document.getElementById('profileForm').classList.remove('hidden');
            document.getElementById('profileDisplay').classList.add('hidden');
        }

        function logoutProfile() {
            currentUser = null;
            extractedResumeData = null;
            localStorage.removeItem('currentUser');
            location.reload();
        }

        // ===== RESUME TAB =====
        function uploadResume() {
            if (!currentUser) {
                showAlert('resumeAlert', 'Please register your profile first!', 'error');
                return;
            }

            const file = document.getElementById('resumeFile').files[0];
            if (!file) {
                showAlert('resumeAlert', 'Please select a resume file', 'error');
                return;
            }

            const formData = new FormData();
            formData.append('file', file);

            fetch(`${API_BASE}/users/${currentUser.id}/upload-resume`, {
                method: 'POST',
                body: formData
            })
            .then(res => {
                if (!res.ok) throw new Error('Upload failed');
                return res.json();
            })
            .then(data => {
                extractedResumeData = data;
                displayResumePreview(data);
            })
            .catch(err => showAlert('resumeAlert', 'Error: ' + err.message, 'error'));
        }

        function displayResumePreview(data) {
            document.getElementById('resumeUploadForm').classList.add('hidden');
            document.getElementById('resumePreview').classList.remove('hidden');

            let previewHtml = '<p>';
            if (data.name) previewHtml += `<strong>Name:</strong> ${data.name}<br>`;
            if (data.email) previewHtml += `<strong>Email:</strong> ${data.email}<br>`;
            if (data.phone) previewHtml += `<strong>Phone:</strong> ${data.phone}<br>`;
            if (data.location) previewHtml += `<strong>Location:</strong> ${data.location}<br>`;
            if (data.yearsExperience) previewHtml += `<strong>Experience:</strong> ${data.yearsExperience} years<br>`;
            if (data.skills) previewHtml += `<strong>Skills:</strong> ${data.skills}<br>`;
            previewHtml += '</p>';

            document.getElementById('previewContent').innerHTML = previewHtml;
        }

        function confirmResume() {
            if (!currentUser || !extractedResumeData) {
                showAlert('resumeAlert', 'Missing data', 'error');
                return;
            }

            fetch(`${API_BASE}/users/${currentUser.id}/confirm-profile`, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(extractedResumeData)
            })
            .then(res => {
                if (!res.ok) throw new Error('Confirmation failed');
                return res.json();
            })
            .then(data => {
                currentUser = data;
                localStorage.setItem('currentUser', JSON.stringify(data));
                showAlert('resumeAlert', 'Resume confirmed and saved!', 'success');
                uploadNewResume();
            })
            .catch(err => showAlert('resumeAlert', 'Error: ' + err.message, 'error'));
        }

        function uploadNewResume() {
            document.getElementById('resumeUploadForm').classList.remove('hidden');
            document.getElementById('resumePreview').classList.add('hidden');
            document.getElementById('resumeFile').value = '';
        }

        // ===== DISCOVERY TAB =====
        function discoverJobs() {
            if (!currentUser) {
                showAlert('discoveryAlert', 'Please register your profile first!', 'error');
                return;
            }

            const request = {
                keywords: document.getElementById('jobKeywords').value || 'Software Engineer',
                location: document.getElementById('jobLocation').value || 'Remote',
                jobType: document.getElementById('jobType').value || ''
            };

            document.getElementById('discoveryLoading').classList.remove('hidden');

            fetch(`${API_BASE}/jobs/discovery/run`, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(request)
            })
            .then(res => {
                if (!res.ok) throw new Error('Discovery failed');
                return res.json();
            })
            .then(data => {
                displayDiscoveryResults(data);
            })
            .catch(err => showAlert('discoveryAlert', 'Error: ' + err.message, 'error'))
            .finally(() => document.getElementById('discoveryLoading').classList.add('hidden'));
        }

        function displayDiscoveryResults(data) {
            const jobs = data.jobs || [];
            let html = '';

            jobs.forEach(job => {
                html += `
                    <div class="job-card">
                        <h3>${job.title || 'Job Title'}</h3>
                        <p><strong>Company:</strong> ${job.company || 'N/A'}</p>
                        <p><strong>Location:</strong> ${job.location || 'Remote'}</p>
                        <p><strong>Description:</strong> ${job.description?.substring(0, 150) || 'N/A'}...</p>
                        <a href="${job.url || '#'}" target="_blank" style="color: #667eea; text-decoration: none;">View Job →</a>
                    </div>
                `;
            });

            document.getElementById('jobsList').innerHTML = html;
            document.getElementById('discoveryResults').classList.remove('hidden');
        }

        // ===== MATCH TAB =====
        function loadMatchedJobs() {
            if (!currentUser) {
                showAlert('matchAlert', 'Please register your profile first!', 'error');
                return;
            }

            document.getElementById('matchLoading').classList.remove('hidden');

            fetch(`${API_BASE}/users/${currentUser.id}`, {
                method: 'GET'
            })
            .then(res => {
                if (!res.ok) throw new Error('Failed to load profile');
                return res.json();
            })
            .then(data => {
                displayMatchedJobs(data);
            })
            .catch(err => showAlert('matchAlert', 'Error: ' + err.message, 'error'))
            .finally(() => document.getElementById('matchLoading').classList.add('hidden'));
        }

        function displayMatchedJobs(data) {
            let html = '<p>No matched jobs yet. Run job discovery first!</p>';
            document.getElementById('matchedJobsList').innerHTML = html;
            document.getElementById('matchResults').classList.remove('hidden');
        }

        // Initialize
        window.addEventListener('load', () => {
            const saved = localStorage.getItem('currentUser');
            if (saved) {
                currentUser = JSON.parse(saved);
                displayProfile();
            }
        });
    </script>

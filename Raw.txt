<script>
const API = '/api';
let user = JSON.parse(localStorage.getItem('jaUser') || 'null');

/* ---------- helpers ---------- */
const $ = id => document.getElementById(id);
const esc = s => String(s ?? '').replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
const list = s => (s || '').split(',').map(x => x.trim()).filter(Boolean);
const num = id => $(id).value === '' ? null : Number($(id).value);
const bool = id => $(id).value === '' ? null : $(id).value === 'true';
const badges = arr => (arr || []).map(x => `<span class="badge">${esc(x)}</span>`).join('') || '-';

function msg(id, text, type = 'info') { $(id).innerHTML = text ? `<div class="msg ${type}">${esc(text)}</div>` : ''; }

async function call(url, opts = {}) {
  const res = await fetch(API + url, opts);
  const text = await res.text();
  let data; try { data = JSON.parse(text); } catch { data = text; }
  if (!res.ok) throw new Error((data && (data.message || data.error)) || `HTTP ${res.status}`);
  return data;
}
const postJson = (url, body) => call(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) });

function requireUser(msgId) {
  if (!user) { msg(msgId, 'Register or load a profile first (Profile tab).', 'err'); return false; }
  return true;
}
async function busy(btnId, fn) {
  const b = $(btnId), t = b.textContent; b.disabled = true; b.textContent = 'Working...';
  try { await fn(); } finally { b.disabled = false; b.textContent = t; }
}

/* ---------- tabs ---------- */
document.querySelectorAll('nav button').forEach(b => b.onclick = () => {
  document.querySelectorAll('nav button').forEach(x => x.classList.remove('active'));
  document.querySelectorAll('section').forEach(x => x.classList.remove('active'));
  b.classList.add('active'); $(b.dataset.tab).classList.add('active');
  if (b.dataset.tab === 'resumes' && user) loadResumes();
});

/* ---------- profile ---------- */
function setUser(u) {
  user = u; localStorage.setItem('jaUser', JSON.stringify(u)); renderProfile();
}
function renderProfile() {
  $('userBadge').textContent = user ? `#${user.id} · ${user.fullName || user.email}` : 'Not logged in';
  $('profileView').classList.toggle('hidden', !user);
  if (!user) return;
  $('profileDetails').innerHTML = `
    <h4>${esc(user.fullName)} (${esc(user.email)})</h4>
    <div class="meta">ID ${user.id} · ${esc(user.currentLocation)} → ${esc(user.preferredLocation)} · ${esc(user.phone)}</div>
    <p><b>Target roles:</b> ${badges(user.targetRoles)}</p>
    <p><b>Past roles:</b> ${badges(user.pastRoles)}</p>
    <p><b>Skills:</b> ${badges(user.skills)}</p>
    <p><b>Experience:</b> ${user.totalExperienceYears ?? '-'} yrs · <b>Education:</b> ${esc(user.education || '-')}</p>
    <p><b>Salary:</b> min ${user.minSalary ?? '-'} / expected ${user.expectedSalary ?? '-'} · <b>Notice:</b> ${user.noticePeriodDays ?? '-'} days</p>
    <p><b>Work authorized:</b> ${user.workAuthorized ?? '-'} · <b>Sponsorship:</b> ${user.requiresSponsorship ?? '-'}</p>
    ${user.profileSummary ? `<p style="margin-top:6px"><i>${esc(user.profileSummary)}</i></p>` : ''}`;
}

async function register() {
  const body = {
    fullName: $('fullName').value, email: $('email').value, phone: $('phone').value,
    linkedinUrl: $('linkedinUrl').value, currentLocation: $('currentLocation').value,
    preferredLocation: $('preferredLocation').value, targetRoles: list($('targetRoles').value),
    minSalary: num('minSalary'), expectedSalary: num('expectedSalary'),
    noticePeriodDays: num('noticePeriodDays'),
    workAuthorized: bool('workAuthorized'), requiresSponsorship: bool('requiresSponsorship')
  };
  try { setUser(await postJson('/users/register', body)); msg('msgReg', `Registered! Your user ID is ${user.id}`, 'ok'); }
  catch (e) { msg('msgReg', 'Registration failed: ' + e.message, 'err'); }
}

async function loadUser() {
  const id = $('loadId').value || user?.id;
  if (!id) return msg('msgLoad', 'Enter a user ID', 'err');
  try { setUser(await call('/users/' + id)); msg('msgLoad', 'Profile loaded', 'ok'); }
  catch (e) { msg('msgLoad', 'Load failed: ' + e.message, 'err'); }
}

function logout() { user = null; localStorage.removeItem('jaUser'); renderProfile(); msg('msgLoad', 'Logged out'); }

/* ---------- resume upload / confirm ---------- */
async function uploadResume() {
  if (!requireUser('msgResume')) return;
  const f = $('resumeFile').files[0];
  if (!f) return msg('msgResume', 'Choose a PDF file', 'err');
  const fd = new FormData(); fd.append('file', f);
  await busy('btnUpload', async () => {
    try {
      msg('msgResume', 'Extracting with AI... (can take ~15s)');
      const ex = await call(`/users/${user.id}/upload-resume`, { method: 'POST', body: fd });
      $('exSkills').value = (ex.skills || []).join(', ');
      $('exRoles').value = (ex.roles || []).join(', ');
      $('exYears').value = ex.totalExperienceYears ?? '';
      $('exEdu').value = ex.education ?? '';
      $('exSummary').value = ex.profileSummary ?? '';
      $('extractBox').classList.remove('hidden');
      msg('msgResume', 'Review and confirm below', 'ok');
    } catch (e) { msg('msgResume', 'Upload failed: ' + e.message, 'err'); }
  });
}

async function confirmProfile() {
  const body = {
    skills: list($('exSkills').value), roles: list($('exRoles').value),
    totalExperienceYears: Math.round(Number($('exYears').value) || 0),
    education: $('exEdu').value, profileSummary: $('exSummary').value
  };
  try {
    setUser(await postJson(`/users/${user.id}/confirm-profile`, body));
    $('extractBox').classList.add('hidden');
    msg('msgResume', 'Profile saved ✅', 'ok');
  } catch (e) { msg('msgResume', 'Confirm failed: ' + e.message, 'err'); }
}

/* ---------- discovery ---------- */
function jobCard(j, extra = '') {
  return `<div class="card">
    <h4>${esc(j.title)}</h4>
    <div class="meta">${esc(j.company)} · ${esc(j.location || '-')} · <span class="badge">${esc(j.source)}</span> ${j.postedAt ? '· ' + esc(j.postedAt) : ''}</div>
    ${extra}
    ${j.jobUrl ? `<a href="${esc(j.jobUrl)}" target="_blank">View job →</a>` : ''}
  </div>`;
}

async function discover() {
  const body = {
    userId: user?.id ?? null,
    keywords: list($('dKeywords').value),
    location: $('dLocation').value || null,
    limitPerSource: num('dLimit')
  };
  await busy('btnDiscover', async () => {
    try {
      msg('msgDiscover', 'Searching job boards...');
      const r = await postJson('/jobs/discovery/run', body);
      msg('msgDiscover', '');
      const counts = Object.entries(r.sourceCounts || {}).map(([k, v]) => `${k}: ${v}`).join(' · ');
      $('discoverSummary').innerHTML = `<h3>${r.totalJobs} jobs found <span class="meta">${esc(counts)}</span></h3>`;
      $('discoverList').innerHTML = (r.jobs || []).map(j => jobCard(j)).join('') || '<p>No jobs found.</p>';
    } catch (e) { msg('msgDiscover', 'Discovery failed: ' + e.message, 'err'); }
  });
}

/* ---------- matching ---------- */
function matchBody() {
  return {
    jobSearchRequest: {
      userId: user.id,
      keywords: list($('mKeywords').value),
      location: $('mLocation').value || null,
      limitPerSource: num('mLimitSrc')
    },
    limit: num('mLimit'),
    minimumScore: num('mMin') ?? 30,
    includeRejected: $('mRejected').value === 'true'
  };
}

async function matchJobs() {
  if (!requireUser('msgMatch')) return;
  await busy('btnMatch', async () => {
    try {
      msg('msgMatch', 'Scoring jobs with AI... (may take a minute)');
      const r = await postJson(`/jobs/match/${user.id}`, matchBody());
      msg('msgMatch', '');
      $('matchSummary').innerHTML = `<h3>${r.totalMatched} matched of ${r.totalJobsEvaluated} evaluated</h3>`;
      $('matchList').innerHTML = (r.matches || []).map(m => jobCard(m.job, `
        <span class="badge ${esc(m.matchLevel)}">${esc(m.matchLevel)} · ${m.matchScore}</span>
        <div class="scores">Keyword: ${m.keywordScore ?? '-'} · AI: ${m.aiScore ?? '-'}</div>
        ${m.aiSummary ? `<p style="font-size:.88em"><i>${esc(m.aiSummary)}</i></p>` : ''}
        ${m.reasons?.length ? `<b style="font-size:.85em">Reasons</b><ul>${m.reasons.map(x => `<li>${esc(x)}</li>`).join('')}</ul>` : ''}
        ${m.concerns?.length ? `<b style="font-size:.85em">Concerns</b><ul>${m.concerns.map(x => `<li>${esc(x)}</li>`).join('')}</ul>` : ''}
      `)).join('') || '<p>No matches above the minimum score.</p>';
    } catch (e) { msg('msgMatch', 'Match failed: ' + e.message, 'err'); }
  });
}

async function notifyHigh() {
  if (!requireUser('msgMatch')) return;
  await busy('btnNotify', async () => {
    try {
      const r = await postJson(`/jobs/match/${user.id}/notify`, matchBody());
      msg('msgMatch', r.status + ` (${user.email})`, 'ok');
    } catch (e) { msg('msgMatch', 'Notify failed: ' + e.message, 'err'); }
  });
}

/* ---------- generated resumes ---------- */
async function loadResumes() {
  if (!requireUser('msgResumes')) return;
  try {
    const rows = await call(`/resumes/user/${user.id}`);
    msg('msgResumes', '');
    $('resumesList').innerHTML = rows.length ? `
      <table><tr><th>ID</th><th>Job</th><th>Company</th><th>Source</th><th>Created</th><th></th></tr>
      ${rows.map(r => `<tr><td>${r.id}</td><td>${esc(r.title)}</td><td>${esc(r.company)}</td>
        <td>${esc(r.source)}</td><td>${esc(r.createdAt)}</td>
        <td><a href="${API}/resumes/${r.id}/pdf">Download PDF</a></td></tr>`).join('')}
      </table>` : '<p>No tailored resumes yet. Use "Email me HIGH matches", then confirm from the email.</p>';
  } catch (e) { msg('msgResumes', 'Load failed: ' + e.message, 'err'); }
}

/* ---------- init ---------- */
renderProfile();
</script>

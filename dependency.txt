# AWS RDS PostgreSQL
spring.datasource.url=jdbc:postgresql://${RDS_HOSTNAME}:${RDS_PORT:5432}/${RDS_DB_NAME:ecommerce_db}
spring.datasource.username=${RDS_USERNAME}
spring.datasource.password=${RDS_PASSWORD}

# Redis — use AWS ElastiCache or keep containerized
spring.data.redis.host=${REDIS_HOST:redis}
spring.data.redis.port=${REDIS_PORT:6379}

# Production logging
logging.level.org.springframework=WARN
logging.level.org.hibernate.SQL=WARN
logging.level.com.ecommerce.project=INFO



-------

services:
  redis:
    image: redis:7-alpine
    container_name: ecom-redis
    ports:
      - "6379:6379"

  app:
    build: .
    container_name: ecom-app
    ports:
      - "8080:8080"
    environment:
      SPRING_PROFILES_ACTIVE: aws
      RDS_HOSTNAME: your-db-instance.xxxxxxxxxxxx.us-east-1.rds.amazonaws.com
      RDS_PORT: 5432
      RDS_DB_NAME: ecommerce_db
      RDS_USERNAME: postgres
      RDS_PASSWORD: ${RDS_PASSWORD}
      REDIS_HOST: redis
      JWT_SECRET: ${JWT_SECRET}
      MAIL_PASSWORD: ${MAIL_PASSWORD}
    depends_on:
      - redis
